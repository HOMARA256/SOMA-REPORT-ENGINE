import {
  ArrowLeft,
  BookOpen,
  CalendarDays,
  Edit3,
  Phone,
  ShieldCheck,
  User,
  Users,
} from "lucide-react";

import {
  useNavigate,
  useParams,
} from "react-router-dom";

import {
  getLearnerFullName,
  getLearnerInitials,
  learners,
} from "./learnerData";

export default function LearnerProfilePage() {
  const navigate = useNavigate();
  const { learnerId } = useParams();

  const learner = learners.find(
    (item) => item.id === learnerId,
  );

  if (!learner) {
    return (
      <section className="teacher-not-found">
        <User size={36} />

        <h1>Learner not found</h1>

        <p>
          The requested learner record could not be found.
        </p>

        <button
          type="button"
          className="primary-button"
          onClick={() => navigate("/people/learners")}
        >
          Return to Learners
        </button>
      </section>
    );
  }

  return (
    <>
      <button
        type="button"
        className="profile-back-button"
        onClick={() => navigate("/people/learners")}
      >
        <ArrowLeft size={16} />
        Back to Learners
      </button>

      <section className="learner-profile-header">
        <div className="learner-profile-identity">
          <div className="learner-profile-avatar">
            {getLearnerInitials(learner)}
          </div>

          <div>
            <span className="eyebrow">
              LEARNER PROFILE
            </span>

            <h1>{getLearnerFullName(learner)}</h1>

            <p>
              {learner.admissionNumber} •{" "}
              {learner.className} {learner.stream}
            </p>

            <div className="teacher-profile-tags">
              <span className="profile-role-badge">
                {learner.programme}
              </span>

              <span
                className={`status-badge ${
                  learner.status === "Active"
                    ? "status-active"
                    : "status-inactive"
                }`}
              >
                {learner.status}
              </span>
            </div>
          </div>
        </div>

        <button
          type="button"
          className="secondary-button"
          onClick={() =>
            navigate(
              `/people/learners/${learner.id}/edit`,
            )
          }
        >
          <Edit3 size={16} />
          Edit Learner
        </button>
      </section>

      <div className="learner-profile-grid">
        <section className="panel">
          <div className="panel-heading">
            <div className="panel-title-group">
              <div className="panel-icon">
                <User size={18} />
              </div>

              <div>
                <h2>Personal Information</h2>
                <p>
                  Learner identity and registration details.
                </p>
              </div>
            </div>
          </div>

          <dl className="profile-details">
            <div>
              <dt>Admission Number</dt>
              <dd>{learner.admissionNumber}</dd>
            </div>

            <div>
              <dt>Gender</dt>
              <dd>{learner.gender}</dd>
            </div>

            <div>
              <dt>Date of Birth</dt>
              <dd>{learner.dateOfBirth}</dd>
            </div>

            <div>
              <dt>Admission Date</dt>
              <dd>{learner.admissionDate}</dd>
            </div>
          </dl>
        </section>

        <section className="panel">
          <div className="panel-heading">
            <div className="panel-title-group">
              <div className="panel-icon">
                <BookOpen size={18} />
              </div>

              <div>
                <h2>Academic Enrolment</h2>
                <p>
                  Current academic placement.
                </p>
              </div>
            </div>
          </div>

          <dl className="profile-details">
            <div>
              <dt>Programme</dt>
              <dd>{learner.programme}</dd>
            </div>

            <div>
              <dt>Class</dt>
              <dd>{learner.className}</dd>
            </div>

            <div>
              <dt>Stream</dt>
              <dd>{learner.stream}</dd>
            </div>

            <div>
              <dt>Status</dt>
              <dd>{learner.status}</dd>
            </div>
          </dl>
        </section>

        <section className="panel">
          <div className="panel-heading">
            <div className="panel-title-group">
              <div className="panel-icon">
                <Users size={18} />
              </div>

              <div>
                <h2>Parent / Guardian</h2>
                <p>
                  Primary guardian contact information.
                </p>
              </div>
            </div>
          </div>

          <dl className="profile-details">
            <div>
              <dt>Guardian</dt>
              <dd>{learner.guardianName}</dd>
            </div>

            <div>
              <dt>Telephone</dt>
              <dd className="profile-muted">
                <Phone size={14} />
                {learner.guardianPhone}
              </dd>
            </div>
          </dl>
        </section>

        <section className="panel">
          <div className="panel-heading">
            <div className="panel-title-group">
              <div className="panel-icon">
                <ShieldCheck size={18} />
              </div>

              <div>
                <h2>Academic Context</h2>
                <p>
                  Current reporting context for this learner.
                </p>
              </div>
            </div>
          </div>

          <div className="learner-context-list">
            <div>
              <CalendarDays size={17} />

              <span>
                <small>Academic Year</small>
                <strong>2026</strong>
              </span>
            </div>

            <div>
              <BookOpen size={17} />

              <span>
                <small>Current Term</small>
                <strong>Term II</strong>
              </span>
            </div>
          </div>
        </section>
      </div>
    </>
  );
}
